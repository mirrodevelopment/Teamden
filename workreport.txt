  function updateLiveDateWidget() {
    const now = new Date();
    const months = ['JAN','FEB','MAR','APR','MAY','JUN','JUL','AUG','SEP','OCT','NOV','DEC'];
    const days = ['SUNDAY','MONDAY','TUESDAY','WEDNESDAY','THURSDAY','FRIDAY','SATURDAY'];
    
    document.getElementById('live-month-year').textContent = `${months[now.getMonth()]} ${now.getFullYear()}`;
    document.getElementById('live-day-number').textContent = String(now.getDate()).padStart(2, '0');
    document.getElementById('live-day-name').textContent = days[now.getDay()];
  }
  setInterval(updateLiveDateWidget, 1000);
  updateLiveDateWidget();

Example A: Unifying Font Architecture into Single Source of Truth

  [BEFORE - 4 Separate Redundant Imports and Conflicting Rules]:
  --------------------------------------------------
  /* In fragments/fonts/fonts.css */
  @import url('../font.css');
  @import url('mostarct.css');
  @import url('inter.css');
  @import url('https://fonts.googleapis.com/css2?family=Inter+Tight...');

  /* In fragments/font.css */
  @import url('https://fonts.googleapis.com/css2?family=Montserrat...');
  body { font-family: 'Mostarct', sans-serif; }

  [AFTER - Unified fragments/fonts/fonts.css with Jx Tabe Black @font-face]:
  --------------------------------------------------
  /* Google Web Font Imports (Must precede all rules) */
  @import url('https://fonts.googleapis.com/css2?family=Montserrat:ital,wght@0,300;0,400;0,500;0,600;0,700;0,800;0,900;1,400&display=swap');
  @import url('https://fonts.googleapis.com/css2?family=Inter:wght@400;500;600;700;800;900&display=swap');
  @import url('https://fonts.googleapis.com/css2?family=Inter+Tight:wght@400;500;600;700;800;900&display=swap');
  @import url('https://fonts.googleapis.com/css2?family=Unbounded:wght@800;900&display=swap');

  /* Universal Number Font: Jx Tabe Black */
  @font-face {
    font-family: 'Jx Tabe Black';
    font-style: normal;
    font-weight: 900;
    font-display: swap;
    src: local('Jx Tabe Black'),
         local('JxTabe-Black'),
         url('/fragments/fonts/JxTabe-Black.woff2') format('woff2'),
         url('/fragments/fonts/JxTabe-Black.woff') format('woff'),
         url('/fragments/fonts/MonumentExtended-Ultrabold.woff') format('woff');
  }

  :root {
    --font-family-base: 'Montserrat', -apple-system, sans-serif;
    --font-family-display: 'Inter', sans-serif;
    --font-family-numbers: 'Jx Tabe Black', 'Monument Extended', sans-serif;
  }

  public ProjectOverviewDto getProjectOverview(String projectId) {
      Project project = resolveProject(projectId);
      User currentUser = resolveCurrentUser(project.getOrganization());
      ensureProjectFlowBaseline(project, currentUser);

      List<ProjectMilestone> milestones = projectMilestoneRepository.findByProjectIdOrderBySeqNumAsc(project.getId());
      Optional<ProjectSpecification> specOpt = projectSpecificationRepository.findByProjectId(project.getId());
      List<Task> tasks = taskRepository.findByProjectIdWithDetails(project.getId());
      List<ProjectDecision> decisions = projectDecisionRepository.findByProjectIdWithDecider(project.getId());
      List<ProjectDocument> docs = projectDocumentRepository.findByProjectIdWithAuthor(project.getId());
      List<ProjectMember> members = projectMemberRepository.findByProjectIdWithUser(project.getId());

      return ProjectOverviewDto.builder()
              .id(project.getId().toString())
              .title(project.getName())
              .milestonesSummary(milestonesSummary)
              .milestonesCompleted(milestonesCompleted)
              .milestonesTotal(milestonesTotal)
